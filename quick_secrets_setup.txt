# Quick GitHub Secrets Setup Commands

# Method 1: Using the automated script (RECOMMENDED)
# Install GitHub CLI first, then run:
./setup_github_secrets.sh

# Method 2: Manual CLI commands (if script doesn't work)
# After installing gh CLI and running 'gh auth login':

source .env

echo "$BASE_URL" | gh secret set BASE_URL
echo "$USERNAME" | gh secret set API_USERNAME
echo "$PASSWORD" | gh secret set API_PASSWORD
echo "$TENANTID" | gh secret set TENANTID
echo "$USERTYPE" | gh secret set USERTYPE
echo "$CLIENT_AUTH_HEADER" | gh secret set CLIENT_AUTH_HEADER
echo "$LOCALE" | gh secret set LOCALE
echo "$LOCALE_FRENCH" | gh secret set LOCALE_FRENCH
echo "$LOCALE_PORTUGUESE" | gh secret set LOCALE_PORTUGUESE
echo "$SEARCH_LIMIT" | gh secret set SEARCH_LIMIT
echo "$SEARCH_OFFSET" | gh secret set SEARCH_OFFSET
